Task 4: Simple Unit Test for Add Function

Function & Test
function add(a, b) {
  return a + b;
}

function testAdd() {
  const result   = add(2, 3);
  const expected = 5;

  if (result === expected) {
    console.log("Test Passed");
  } else {
    console.log(`Test Failed: expected ${expected}, got ${result}`);
  }
}

testAdd();
