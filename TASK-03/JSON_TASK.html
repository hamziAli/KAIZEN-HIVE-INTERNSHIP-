Task 2: Display JSON Array as List
JavaScript Code
const jsonArray = '["HTML", "CSS", "JavaScript"]';
const courses = JSON.parse(jsonArray);

const ul = document.createElement("ul");
courses.forEach(course => {
  const li = document.createElement("li");
  li.textContent = course;
  ul.appendChild(li);
});
document.getElementById("courseList").appendChild(ul);
Output

    Task 4: Search Item in JSON Data
    JavaScript Code
    const students = [
  {"roll": "CS1", "name": "Ali"},
  {"roll": "CS2", "name": "Sara"}
];
const searchRoll = "CS2";

const found = students.find(s => s.roll === searchRoll);

  console.log("Student Name: " + found.name);
} else {
  console.log("Student not found.");
