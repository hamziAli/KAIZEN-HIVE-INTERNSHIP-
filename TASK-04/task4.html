Task 4: Grade Calculator
JavaScript Code
const marks = 76;
let grade;

if (marks >= 80) {
  grade = "A";
} else if (marks >= 60) {
  grade = "B";
} else if (marks >= 40) {
  grade = "C";
} else {
  grade = "Fail";
}

console.log("Grade: " + grade);
Output
